import struct
import sys
sys.path.insert(0, 'C:/EnBDevelopment/EnB/.work/a13')
import emu  # noqa: E402
from harness import Harness, fhash  # noqa: E402

emu.ALIASES['?Unknown_468BE0@LandUnknown466BE0Class@@QAEXAAVAABoxClass@@PBVVector3@@@Z'] = 0x468BE0
OBJ = 'C:/EnBDevelopment/EnB/.work/a13/objs/landdmap.obj'
SYM = '?Unknown_468920@LandUnknown466BE0Class@@QAEXXZ'
THIS, VBUF, VTX, HEAP = 0x30000000, 0x31000000, 0x50000000, 0x60000000
DELETE, NEW, BOX = 0x4E9D10, 0x4E9CB0, 0x468BE0
CASES = [(n, nv, old, newnull) for n in (0, 1, 2, 3, 5) for nv in (9, 25) for old in (0, 1) for newnull in (0, 1)]


def setup(case):
    n, nv, old, newnull = case

    def f(mem, cpu, seed):
        mem.wu(THIS + 4, 4, 0x61000000 if old else 0)
        mem.wu(THIS + 0x14, 4, n)
        mem.wu(THIS + 0x24, 4, VBUF)
        mem.wu(VBUF, 4, 0x5C000000)
        mem.wu(THIS + 0x3C, 4, nv)
        for k in range(nv):
            mem.write(VTX + 0x34 * k, struct.pack('<3f', fhash(seed, k, 1) * 9, fhash(seed, k, 2) * 9,
                                                   fhash(seed, k, 3)))
        cpu.r['ecx'] = THIS
        return []
    return f


def stubs(case):
    n, nv, old, newnull = case

    def f(cpu, key, words, idx, seed):
        m = cpu.m
        if key == ('direct', DELETE):
            return 0, {'_cdecl': 1}, 1
        if key == ('direct', NEW):
            cpu.r['eax'] = 0 if newnull else HEAP
            return 0, {'_cdecl': 1}, 1
        if key[0] == 'indirect' and key[1] == 0x2C:
            m.wu(words[3], 4, VTX)
            return 20, {'pp': 'stk' if emu.STACK_LO <= words[3] < emu.STACK_TOP else words[3]}, 3
        if key[0] == 'indirect' and key[1] == 0x30:
            return 4, None, 1
        if key == ('direct', BOX):
            return 8, {'rec': words[0], 'pts': m.read(words[1], 96).hex()}, 0
        raise emu.Stop('callee %r' % (key,))
    return f


def main():
    H = Harness(OBJ, SYM, 0x468920, 0, void=True, stk_any=True)
    if len(sys.argv) > 1 and sys.argv[1] == 'ctl':
        base = [(3, 9, 1, 0), (2, 25, 0, 0), (5, 25, 1, 0)]
        H.run(base, setup, stubs, seeds=(1,), label='base')
        off, ins = H.find(lambda i: i.mnemonic == 'cmp' and i.op_str.endswith(', 4'))
        H.mutate('1 (+%#x %s %s: 4 corners -> 3)' % (off, ins.mnemonic, ins.op_str), off, ins.size - 1, 3,
                 base, setup, stubs, seeds=(1,))
        off, ins = H.find(lambda i: i.mnemonic == 'fadd' and '0x21' in i.op_str)
        H.mutate('2 (+%#x %s %s: the raised +1.0 -> another constant)' % (off, ins.mnemonic, ins.op_str), off,
                 ins.size - 4, (H.reader.code[off + ins.size - 4] + 0x10) & 0xFF, base, setup, stubs, seeds=(1,))
        off, ins = H.find(lambda i: i.op_str == 'eax, dword ptr [edi + eax + 0x24]')
        H.mutate('3 (+%#x %s %s: corner 3 index read +0x24 -> +0x20)' % (off, ins.mnemonic, ins.op_str), off,
                 ins.size - 1, 0x20, base, setup, stubs, seeds=(1,))
        return
    H.run(CASES, setup, stubs, seeds=(1, 2), verbose=False)
    H.coverage()


if __name__ == '__main__':
    main()
