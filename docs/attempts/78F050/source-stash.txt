def once(s, old, new):
    assert s.count(old) == 1, (old, s.count(old))
    return s.replace(old, new, 1)

def save(name, value):
    # This TU has padding after the target. Preserve every following physical
    # line, especially FAIL_IF line 708, without preprocessor line directives.
    available = base[begin:end].count('\n')
    assert value.count('\n') <= available
    source = base[:begin] + value + '\n' * (available-value.count('\n')) + base[end:]
    assertions = lambda s: [(i, line.strip()) for i,line in enumerate(s.splitlines(),1)
                             if re.search(r'\b(?:G_ASSERT|FAIL_IF)\s*\(',line)]
    assert assertions(source) == assertions(base)
    assert '#line' not in source
    folder = OUT / name
    folder.mkdir(exist_ok=True)
    (folder / 'nlp.cpp').write_text(source)
    return {'name': name, 'source': str(folder / 'nlp.cpp')}

def byte_locals(s):
    lines = []
    for line in s.splitlines(True):
        term = '(unsigned char)Unknown_78EEF0(start, count)'
        if term in line:
            indent = line[:len(line)-len(line.lstrip())]
            lines.append(indent + 'unsigned char animation = '+term+'; // NAME 3: client AL byte spill before stream call\n')
            line = line.replace(term, 'animation')
        lines.append(line)
    return ''.join(lines)

def phrase_order(s):
    old = '''\t\t\t\t\tUnknown_78F700Class::BAnimStruct *gesture = Personality->Find_BAnim(tag);
\t\t\t\t\tif (count + n != 0) {

\t\t\t\t\tcount = 0;
\t\t\t\t\tp += n;
\t\t\t\t\tstart = p;'''
    new = '''\t\t\t\t\tp += n;
\t\t\t\t\tUnknown_78F700Class::BAnimStruct *gesture = Personality->Find_BAnim(tag);
\t\t\t\t\tcount += n;
\t\t\t\t\tif (count != 0) {
\t\t\t\t\t\tstream->Unknown_4C45B0(1, (unsigned char)gesture->AnimNumber, start, count);
\t\t\t\t\t}
\t\t\t\t\tcount = 0;
\t\t\t\t\tstart = p;'''
    return once(s, old, new)

def punct_order(s):
    return once(s,
        '\t\t\t\tstream->Unknown_4C45B0(1, animation, phrase, count + 1);',
        '\t\t\t\tcount++;\n\t\t\t\tstream->Unknown_4C45B0(1, animation, phrase, count);')

def mood_positive(s):
    start = s.index('\t\t\t\tif (number < 0) {')
    stop = s.index('\n\t\t\t} else {',start)
    old = s[start:stop]
    early = '''\t\t\t\tif (number < 0) {
\t\t\t\t\tp += cmdlen;
\t\t\t\t\tcount += cmdlen;
\t\t\t\t\tcontinue;
\t\t\t\t}
'''
    assert old.startswith(early)
    tail = old[len(early):]
    new = '\t\t\t\tif (number >= 0) {\n' + ''.join('\t'+line for line in tail.splitlines(True)) + '\n\t\t\t\t} else {\n\t\t\t\t\tp += cmdlen;\n\t\t\t\t\tcount += cmdlen;\n\t\t\t\t\tcontinue;\n\t\t\t\t}'
    return s[:start] + new + s[stop:]
