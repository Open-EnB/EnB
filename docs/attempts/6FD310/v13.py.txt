BLANK = '}\n\n\n// 0x6FD2C0\n'
assert src.count(BLANK) == 1
AT = 'DeltaArrayClass<UnknownB01A40Type>'
FINDC = ('static inline ' + AT + ' *F8_Find_Changed(DeltaCompoundClass *root, const NameClass &name) '
         '{ ' + AT + ' *t = Unknown_Find_Field<' + AT + ' >(root, name); if (!t) return(NULL); '

    old = src[lo:hi]
    n = old.count('\n')
    assert text.count('\n') <= n, (text.count('\n'), n)
    return (C, old, text + '\n' * (n - text.count('\n')))
