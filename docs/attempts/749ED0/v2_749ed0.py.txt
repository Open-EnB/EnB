NLINES = FUNC.count('\n')

CHANNELS = [

SEND_PART = FUNC[FUNC.index('\tchar buffer[512];\n'):]

def pad(body):
    cur = body.count('\n')
    assert cur <= NLINES, (cur, NLINES)
    idx = body.rindex('}\n')
    return [(FUNC, body[:idx] + '\n' * (NLINES - cur) + body[idx:])]

def build_check_in_enf(rad_and=False, use_done=False, done_at_end=False, rad_pos4=False,
                       words_done=False, msg_done=False, self_rad=False,
                       channel_assign_op=False, diag_d=0, no_goto_check=False,
                       rad_check_goto=False, invalid_in_else=True, send_goto=False):
    lines = ['bool SClient::Unknown_749ED0(const char *text)', '{']
    if diag_d > 0:
        decls = ', '.join(f'_d{k}=0' for k in range(diag_d))
        lines.append(f'\tint {decls};')
    lines.append('\tOptionClass *options = OptionClass::Instance;')
    lines.append('\tstd::string channel = "";')
    rt = 'goto done;' if use_done else 'return(true);'
    for i, (cmd, o, ch) in enumerate(CHANNELS):
        kw = 'if' if i == 0 else '} else if'
        lines.append(f'\t{kw} (strncmp(text, "{cmd}", 4) == 0) {{')
        if i < len(CHANNELS) - 1:
            if no_goto_check:
                lines.append(f'\t\tif (text[4] != \'\\0\' && text[4] != \' \') goto invalid; if (options->Get_Bool("{o}")) channel.assign("{ch}");')
            else:
                lines.append(f'\t\tif (text[4] != \'\\0\' && text[4] != \' \') goto invalid; if (options->Get_Bool("{o}")) channel.assign("{ch}"); goto check_chan;')
        else:
            lines.append(f'\t\tif (text[4] != \'\\0\' && text[4] != \' \') goto invalid; if (options->Get_Bool("{o}")) channel.assign("{ch}");')
            if not no_goto_check:
                lines.append('check_chan:')
                lines.append(f'\t\tif (channel.length() == 0) {{ Unknown_74D990(0x11, "Not monitoring that channel.", 0); {rt} }}')
    if rad_and:
        lines.append('\t} else if (strncmp(text, "/rad", 4) == 0 && (text[4] == \'\\0\' || text[4] == \' \')) {')
    else:
        lines.append('\t} else if (strncmp(text, "/rad", 4) == 0) {')
        if not rad_pos4:
            lines.append('\t\tif (text[4] != \'\\0\' && text[4] != \' \') goto invalid;')
    if rad_pos4:
        lines.append('\t\tif (text[4] == \'\\0\' || text[4] == \' \') {')
    if channel_assign_op:
        lines.append('\t\tchannel = ChatPanel->Unknown_5682C0();')
    else:
        lines.append('\t\tchannel.assign(ChatPanel->Unknown_5682C0());')
    if self_rad:
        lines.append(f'\t\tif (channel.length() == 0) {{ ((SClient *volatile)this)->Unknown_74D990(0x11, "Not monitoring a private channel.", 0); {rt} }}')
    elif rad_check_goto:
        lines.append(f'\t\tif (channel.length() != 0) goto send; Unknown_74D990(0x11, "Not monitoring a private channel.", 0); {rt}')
    else:
        lines.append(f'\t\tif (channel.length() == 0) {{ Unknown_74D990(0x11, "Not monitoring a private channel.", 0); {rt} }}')
    if send_goto and not rad_check_goto:
        lines.append('\t\tgoto send;')
    if rad_pos4:
        lines.append('\t\t} else goto invalid;')
    if invalid_in_else:
        lines.append('\t} else {')
        lines.append('invalid:')
        lines.append('\t\treturn(false);')
        lines.append('\t}')
    else:
        lines.append('\t} else goto invalid;')
    if use_done and not done_at_end:
        lines.append('done: return(true);')
    if send_goto or rad_check_goto:
        lines.append('send:')
    send = SEND_PART
    if words_done:
        send = send.replace('\t\tUnknown_74D990(0x11, "No message to send.", 0);\n\t\treturn(true);\n',
                            '\t\tUnknown_74D990(0x11, "No message to send.", 0);\n\t\tgoto done;\n', 1)
    if msg_done:
        send = send.replace('\t\treturn(true);\n', '\t\tgoto done;\n')
    if done_at_end:
        send = send.replace('\treturn(true);\n}\n', 'done:\n\treturn(true);\n}\n')
    if not invalid_in_else:
        send = send.replace('}\n', 'invalid: return(false);\n}\n')
    body = '\n'.join(lines) + '\n' + send
    return pad(body)

# Also test: base structure but with check_chan in /enf + done at end, and with words declared before buffer
def build_words_order(words_first=False, msg_first=False, buf_top=False):
    t = FUNC
    if words_first:
        t = t.replace(
            '\tchar buffer[512];\n\tBounded_Strcpy(buffer, 512, text);\n\tstd::vector<std::string> words;\n',
            '\tstd::vector<std::string> words;\n\tchar buffer[512];\n\tBounded_Strcpy(buffer, 512, text);\n'
        )
    if buf_top:
        t = t.replace('\tOptionClass *options = OptionClass::Instance;\n', '\tOptionClass *options = OptionClass::Instance; char buffer[512];\n')
        t = t.replace('\tchar buffer[512];\n', '\t\n')
    assert t.count('\n') == NLINES
    return [(FUNC, t)]
