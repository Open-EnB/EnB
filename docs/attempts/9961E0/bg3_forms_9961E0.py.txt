# ==== .work/bg3/v2.py
import runpy
A = runpy.run_path(r'C:\EnBDevelopment\EnB\.work\f4\vdist_a.py')
B = runpy.run_path(r'C:\EnBDevelopment\EnB\.work\f4\vdist_b.py')
D = runpy.run_path(r'C:\EnBDevelopment\EnB\.work\f4\vdist_d.py')
U, P2, LOOPS = A['U'], A['P2'], A['LOOPS']
RUN_INC, DEAD_END, C1 = B['RUN_INC'], B['DEAD_END'], D['C1']


def run(adds):
    return (U, LOOPS, "\tint row1 = 1;\n\tfor (j = 0; j < ny - 1; j++) {\n\t\tfor (i = 0; i < nx - 1; i++) {\n"
            + ''.join("\t\t\tindices.Add(%s);\n" % x for x in adds))


# H1 (BG3 form 1): the client does not CSE Adds 3 and 4 (each computes lea ecx, [ebx + ebp] at its store) while
# it does CSE Adds 2 and 5 (esi): the 3rd and 4th are different expressions of the same value. Written with
# the operands swapped in the 4th, row1 keeps two uses in the inner loop and should win ebp (the client's).
ADDS_H1 = ['j * nx + i', '(j + 1) * nx + i', 'row1 + i', 'i + row1', '(j + 1) * nx + i', '(j + 1) * nx + i + 1']
V = {
    'h1': [P2, C1, run(ADDS_H1), RUN_INC, DEAD_END],
}

import runpy
A = runpy.run_path(r'C:\EnBDevelopment\EnB\.work\f4\vdist_a.py')
B = runpy.run_path(r'C:\EnBDevelopment\EnB\.work\f4\vdist_b.py')

V4 = runpy.run_path(r'C:\EnBDevelopment\EnB\.work\wt\BG3\.work\bg3\v4.py')
U, P2, LOOPS = A['U'], A['P2'], A['LOOPS']
DEAD_END, C1, ADDS6 = B['DEAD_END'], D['C1'], B['ADDS6']
SET = V4['SET']

# H4 (BG3 form 4): the outer index loop's step. The client increments j before the running row base
# (inc edx; add ebp, ecx; cmp edx, esi), ours (row1 += nx at the end of the body) the reverse: the base is
# stepped in the for statement after j, `for (j = 0; j < ny - 1; j++, row1 += nx)`. A loop-step variable
# may also change its allocation weight (the client keeps it in ebp, ours spills it).
RUN_STEP = (U, LOOPS, "\tint row1 = 1;\n\tfor (j = 0; j < ny - 1; j++, row1 += nx) {\n\t\tfor (i = 0; i < nx - 1; i++) {\n"
            + ''.join("\t\t\tindices.Add(%s);\n" % x for x in ADDS6))
V = {
    'h4': [P2, C1, RUN_STEP, DEAD_END, SET],
}

import runpy
D = runpy.run_path(r'C:\EnBDevelopment\EnB\.work\f4\vdist_d.py')

# F2 (BG3 form 5): H2 showed the client's frame (0x208, the accessors at 0x1ac / 0x1b8 shared with nothing) once
# the accessors live as long as the function, but their destructors then ran after the three releases. The
# client runs them before (~IB, ~VB, surface, texture, back buffer): the accessors' scope is a block that ends
# before the releases yet starts at the top, i.e. the body after the Unknown8C test is one block, the three
# pointers released after it are declared before it, and the accessors sit in that block directly.
TOP = (U, "\tif (!Unknown8C) return;\n",
       "\tif (!Unknown8C) return;\n\tSurfaceClass * back_buffer;\n\tTextureClass * texture;\n\tSurfaceClass * surface;\n\t{\n")
BB = (U, "\tSurfaceClass * back_buffer = DX8Wrapper::", "\tback_buffer = DX8Wrapper::")
TX = (U, "\tTextureClass * texture = NEW_REF(", "\ttexture = NEW_REF(")
SF = (U, "\tSurfaceClass * surface = texture->", "\tsurface = texture->")
OPEN = (U, "\t{\n\t\tDynamicVBAccessClass vb_access(", "\t\tDynamicVBAccessClass vb_access(")
V = {
    'f2': RD_C1 + [SET, TOP, BB, TX, SF, OPEN],
}
