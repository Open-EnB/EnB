         "\t\t\tint i0 = j * nx + i;\n\t\t\tint i1 = (j + 1) * nx + i;\n\t\t\tindices.Add(i0);\n\t\t\tindices.Add(i1);\n"

    return (U, LOOPS, head + pre + ''.join("\t\t\tindices.Add(%s);\n" % x for x in adds))

TAIL = "\tback_buffer->Release_Ref();\n}"
DEAD_END = (U, TAIL, "\tback_buffer->Release_Ref();\n\tint dead = indices.Count();\n}")
# The client's loop: i live in ebx, j*nx (spilled) and (j+1)*nx (edi) by imul per row, and i + j*nx + 1 through
# ebp, a value that starts at 1 and grows by nx per row (add ebp, ecx) -- a running row variable in the source
# or VC6's strength reduction of the hoisted j*nx + 1; the fa/fb forms have the structure (i live, imuls) but
# no such induction variable, and they drop the 6th ??_H temporary (budget)
ADDS6 = ['j * nx + i', '(j + 1) * nx + i', 'row1 + i', 'row1 + i', '(j + 1) * nx + i', '(j + 1) * nx + i + 1']

       + ''.join("\t\t\tindices.Add(%s);\n" % x for x in ADDS6))
RUN_INC = (U, "\t\t\tindices.Add((j + 1) * nx + i + 1);\n\t\t}\n\t}\n", "\t\t\tindices.Add((j + 1) * nx + i + 1);\n\t\t}\n\t\trow1 += nx;\n\t}\n")

        + ''.join("\t\t\tindices.Add(%s);\n" % x for x in ['row0 + i', 'row1 + i', 'row0 + 1 + i', 'row0 + 1 + i', 'row1 + i', 'row1 + 1 + i']))
V = {
    # the budget: fa plus a dead inline read at the end (W3's M2 lever) to bring the 6th temporary back
    'fa_p2_dead': [P2, F_A, DEAD_END],
    # an explicit running row variable (the client's ebp), the rest as fa
    'run_p2': [P2, RUN, RUN_INC],
    # per-row locals with the +1 grouped with the row base
    'fb1_p2': [P2, F_B1],
}
