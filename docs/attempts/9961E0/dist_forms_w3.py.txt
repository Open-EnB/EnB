LIB = 'ww3d2'
VIEW = "\t\tDX8Wrapper::Set_Transform(D3DTS_VIEW, view);\n"
P2 = (VIEW, "\t\tDX8Wrapper::Set_Transform(D3DTS_VIEW, Matrix4(view));\n")
LOOPS = ("\tfor (j = 0; j < ny - 1; j++) {\n\t\tfor (i = 0; i < nx - 1; i++) {\n"
         "\t\t\tint i0 = j * nx + i;\n\t\t\tint i1 = (j + 1) * nx + i;\n\t\t\tindices.Add(i0);\n\t\t\tindices.Add(i1);\n"
         "\t\t\tindices.Add(i0 + 1);\n\t\t\tindices.Add(i0 + 1);\n\t\t\tindices.Add(i1);\n\t\t\tindices.Add(i1 + 1);\n")
def body(pre, a):
    return [P2, (LOOPS, "\tfor (j = 0; j < ny - 1; j++) {\n\t\tfor (i = 0; i < nx - 1; i++) {\n" + pre +
                 ''.join("\t\t\tindices.Add(%s);\n" % x for x in a))]
FORMS = {
    'base2': [P2],
    # client: i0 = i + j*nx (imul each pass), i1 = i + (j+1)*nx, the third/fourth reduced as i + (j*nx + 1)
    'Q1': body('', ['i + j * nx', 'i + (j + 1) * nx', 'i + j * nx + 1', 'i + j * nx + 1', 'i + (j + 1) * nx', 'i + (j + 1) * nx + 1']),

    'Q3': body('\t\t\tint i0 = i + j * nx;\n\t\t\tint i1 = i + (j + 1) * nx;\n', ['i0', 'i1', 'i0 + 1', 'i0 + 1', 'i1', 'i1 + 1']),
    'Q4': body('\t\t\tunsigned short i0 = j * nx + i;\n\t\t\tunsigned short i1 = (j + 1) * nx + i;\n', ['i0', 'i1', 'i0 + 1', 'i0 + 1', 'i1', 'i1 + 1']),
    'Q5': body('\t\t\tint i0 = j * nx + i;\n\t\t\tint i1 = i0 + nx;\n', ['i0', 'i1', 'i0 + 1', 'i0 + 1', 'i1', 'i1 + 1']),
    'Q6': body('', ['(j * nx) + i', '((j + 1) * nx) + i', '(j * nx) + (i + 1)', '(j * nx) + (i + 1)', '((j + 1) * nx) + i', '((j + 1) * nx) + (i + 1)']),
