# ---- front.py ----
"""Per form of a trybatch batch: the differing rows of one client address window, plus the count of
differing rows outside it (a budget / shape flip shows there).

    python front.py BATCHDIR OBJNAME SYMBOL ADDR LO HI [--full FORM]

Rows whose only difference is a masked jump target or relocation (`REL` on the client side, a
jump on both sides) are not counted.
"""
import os
import re
import subprocess
import sys

ROOT = 'C:/EnBDevelopment/EnB/.work/wt/FB1'
JMP = re.compile(r'^(j[a-z]+|call|push|mov|or|cmp|lea|fld|fmul|fadd|fsub|fdiv|fcomp|fstp|fst|and|add|sub)\b')


def rows(obj, symbol, addr):
    out = subprocess.run([sys.executable, ROOT + '/tools/re/objdiff.py', obj, symbol, addr],
                         capture_output=True, text=True).stdout.splitlines()
    res = []
    for line in out[1:]:
        if '|' not in line:
            continue
        left, right = line.split('|', 1)
        star = left.startswith('*')
        l = left[1:].strip()
        r = right.strip()
        lt = l.split(' ', 1)[1] if ' ' in l else ''
        ra, rt = (r.split(' ', 1) + [''])[:2] if r else ('', '')
        res.append((star, lt.strip(), ra, rt.strip()))
    return res


def real(lt, rt):
    if not lt or not rt:
        return True
    if 'REL' in rt or 'REL' in lt:
        a = lt.split(' ')[0]
        b = rt.split(' ')[0]
        if a == b:
            return False
    return lt != rt


def main():
    batch, objname, symbol, addr, lo, hi = sys.argv[1:7]
    lo = int(lo, 16)
    hi = int(hi, 16)
    full = sys.argv[8] if len(sys.argv) > 8 and sys.argv[7] == '--full' else None
    seen = {}
    for form in sorted(os.listdir(batch)):
        obj = os.path.join(batch, form, 'obj', objname)
        if not os.path.exists(obj):
            continue
        rs = rows(obj, symbol, addr)
        inside = []
        outside = 0
        last = 0
        for star, lt, ra, rt in rs:
            if ra:
                try:
                    last = int(ra, 16)
                except ValueError:
                    pass
            if not star or not real(lt, rt):
                if full == form and lo <= last < hi:
                    inside.append('    %-38s | %s %s' % (lt, ra, rt))
                continue
            if lo <= last < hi:
                inside.append('  * %-38s | %s %s' % (lt, ra, rt))
            else:
                outside += 1
        key = '\n'.join(inside) + '|%d' % outside
        if key in seen and full != form:
            print('%s: = %s' % (form, seen[key]))
            continue
        seen[key] = form
        print('%s: outside %d, inside %d' % (form, outside, len([x for x in inside if x.startswith('  *')])))
        for x in inside[:40]:
            print(x)


main()
