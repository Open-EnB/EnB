def wrap_slice(a, b, extra=None):

    body_lines = []
    for ln in sub:
        s = ln.strip()
        for sym in ('Unknown444', 'Unknown2F4', 'Unknown488', 'Colors', 'Unknown494', 'Unknown4B4', 'Unknown4A4',
                    'Unknown303', 'Unknown26C', 'Unknown270', 'Unknown274', 'Unknown254'):

        body_lines.append(s)
    helper = 'static inline void Reset_Part(FontSys *f) { ' + ' '.join(body_lines) + ' } ' + OLD_PRINT_FN
    new_lines = LINES[:a] + ['\t\t\t\tReset_Part(this);'] + [''] * (b - a - 1) + LINES[b:]

    edits = [(OLD_PRINT_FN, helper), (OLD_UNK1_BODY, '\n'.join(new_lines))]
    if extra:
        edits.extend(extra)
    return edits
