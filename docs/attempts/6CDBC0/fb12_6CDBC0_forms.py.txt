def body(lines, f='f->'):
    out = []
    for l in lines:
        l = l.strip()
        for m in ('Unknown274', 'Unknown254', 'Unknown26C', 'Unknown270', 'Unknown303'):
            l = l.replace(m, f + m)
        out.append(l)
    return ' '.join(out)


def helper(name, lines, kind='static inline'):
    return '%s void %s(FontSys *f) { %s }' % (kind, name, body(lines))


def form(groups, kind='static inline'):
    """groups: list of (name, lines) moved into helpers; every moved line is replaced, the count kept"""
    hunks = []
    defs = []
    for name, lines in groups:
        old = ''.join(lines)
        new = T4 + name + '(this);\n' + '\n' * (len(lines) - 1)
        hunks.append((old, new))
        defs.append(helper(name, lines, kind))
    hunks.append((ANCHOR, ' '.join(defs) + '\n'))
    return hunks
