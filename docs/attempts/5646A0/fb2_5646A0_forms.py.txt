# ==== sib.py (reads every inlined _Eos row of a batch's objects) ====
"""Per form of a trybatch batch: every `mov byte ptr [reg + reg], 0` of our function with the two rows before it
(the inlined _Eos sites), so a form is read apart from the budget / layout state.

    python sib.py BATCHDIR OBJNAME SYMBOL ADDR [PATTERN]
"""
import os
import re
import subprocess
import sys

ROOT = 'C:/EnBDevelopment/EnB/.work/wt/FB2'
PAT = re.compile(sys.argv[5] if len(sys.argv) > 5 else r'mov byte ptr \[e.. \+ e..\], 0')


def rows(obj, symbol, addr):
    out = subprocess.run([sys.executable, ROOT + '/tools/re/objdiff.py', obj, symbol, addr],
                         capture_output=True, text=True).stdout.splitlines()
    res = []
    for line in out[1:]:
        if '|' not in line:
            continue
        left = line.split('|', 1)[0]
        l = left[1:].strip()
        lt = l.split(' ', 1)[1].strip() if ' ' in l else ''
        if lt:
            res.append(lt)
    return res


def main():
    batch, objname, symbol, addr = sys.argv[1:5]
    seen = {}
    for form in sorted(os.listdir(batch)):
        obj = os.path.join(batch, form, 'obj', objname)
        if not os.path.exists(obj):
            continue
        rs = rows(obj, symbol, addr)
        hits = []
        for i, t in enumerate(rs):
            if PAT.search(t):
                hits.append(' ; '.join(rs[max(0, i - 2):i + 2]))
        key = '\n'.join(hits)
        if key in seen:
            print('%s: = %s' % (form, seen[key]))
            continue
        seen[key] = form
        print('%s: %d rows' % (form, len(rs)))
        for h in hits:
            print('    ' + h)


main()
