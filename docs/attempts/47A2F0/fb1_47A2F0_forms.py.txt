# ---- count47.py ----
"""Per form of a trybatch batch of Build_Decorations: real differing rows by client window.

    python count47.py BATCHDIR

Windows: pre (before 0x47B3C4), sx (s * axis, 0x47B3C4-0x47B40A), rv (the three Rotate_Vector, 0x47B40A-0x47B79F),
rec (the record loop and the rest, from 0x47B79F). A row whose only difference is a masked relocation or jump target
is not counted.
"""
import os
import subprocess
import sys

ROOT = 'C:/EnBDevelopment/EnB/.work/wt/FB1'
WIN = [('pre', 0x47A2F0, 0x47B3C8), ('sx', 0x47B3C8, 0x47B40A), ('rv', 0x47B40A, 0x47B79F), ('rec', 0x47B79F, 0x47C580)]


def real(lt, rt):
    if not lt or not rt:
        return True
    if 'REL' in rt or 'REL' in lt or lt.startswith('j') and rt.startswith('j'):
        if lt.split(' ')[0] == rt.split(' ')[0]:
            return False
    return lt != rt


def main():
    batch = sys.argv[1]
    seen = {}
    for form in sorted(os.listdir(batch)):
        obj = os.path.join(batch, form, 'obj', 'unknown_474530.obj')
        if not os.path.exists(obj):
            continue
        out = subprocess.run([sys.executable, ROOT + '/tools/re/objdiff.py', obj, 'Build_Decorations', '0x47A2F0'],
                             capture_output=True, text=True).stdout.splitlines()
        counts = {w[0]: 0 for w in WIN}
        last = 0x47A2F0
        for line in out[1:]:
            if '|' not in line:
                continue
            left, right = line.split('|', 1)
            r = right.strip()
            if r:
                try:
                    last = int(r.split(' ', 1)[0], 16)
                except ValueError:
                    pass
            if not left.startswith('*'):
                continue
            l = left[1:].strip()
            lt = l.split(' ', 1)[1].strip() if ' ' in l else ''
            rt = r.split(' ', 1)[1].strip() if ' ' in r else ''
            if not real(lt, rt):
                continue
            for name, lo, hi in WIN:
                if lo <= last < hi:
                    counts[name] += 1
        key = tuple(counts[w[0]] for w in WIN)
        tag = ''
        if key in seen:
            tag = '  (as %s)' % seen[key]
        else:
            seen[key] = form
        print('%-22s pre %3d  sx %2d  rv %3d  rec %3d%s' % (form, key[0], key[1], key[2], key[3], tag))


main()
