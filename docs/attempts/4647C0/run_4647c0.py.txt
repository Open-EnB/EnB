"""A13: ours vs the client's HeightFieldClass::Cast_Ray (grid walk with two triangle tests per cell) in emu.py."""
import math
import struct
import sys
sys.path.insert(0, 'C:/EnBDevelopment/EnB/.work/a13')
import emu  # noqa: E402
from harness import Harness, fhash, f32  # noqa: E402

OBJ = 'C:/EnBDevelopment/EnB/.work/a13/objs/heightfield.obj'
SYM = '?Cast_Ray@HeightFieldClass@@QAE_NAAVRayCollisionTestClass@@@Z'
THIS, HT, RAY, RES = 0x30000000, 0x60000000, 0x40000000, 0x41000000
GETH, CEIL, FLOOR, INVSQRT, COLLIDE = 0x464600, 0xA27827, 0xA26D75, 0x447C40, 0x8C91E0
RAYS = ('diag', 'xonly', 'yonly', 'vertical', 'outside', 'partial', 'long', 'rev', 'startbad', 'tiny', 'vx_out', 'hy_out', 'before', 'after')
CASES = [(lg, sx, wrap, ray, hitp) for lg in (3, 4) for sx in (1.0, 2.5) for wrap in (0, 1) for ray in RAYS
         for hitp in (0, 3, 7)]


def setup(case):
    lg, sx, wrap, ray, hitp = case
    n = 1 << lg

    def f(mem, cpu, seed):
        mem.wu(THIS + 0, 4, HT)
        mem.write(THIS + 0xC, struct.pack('<3f', sx, sx * 0.8, 0.01))
        mem.wu(THIS + 0x18, 1, wrap)
        mem.wu(THIS + 7 * 4, 4, lg)
        mem.wu(THIS + 8 * 4, 4, n)
        mem.wu(THIS + 10 * 4, 4, n - 1)
        mem.wu(THIS + 12 * 4, 4, n // 2)
        for k in range(n * n):
            mem.wu(HT + 2 * k, 2, int((fhash(seed, k) + 4) * 300))
        ext = n / 2 * sx
        r = lambda k: fhash(seed, 40 + k) / 4 * ext  # noqa: E731
        p0 = [r(0), r(1), 50.0]
        p1 = [r(2), r(3), -5.0]
        if ray == 'xonly':
            p1[1] = p0[1]
        elif ray == 'yonly':
            p1[0] = p0[0]
        elif ray == 'vertical':
            p1[0], p1[1] = p0[0], p0[1]
        elif ray == 'outside':
            p0[0] += 3 * ext
            p1[0] += 3 * ext
        elif ray == 'partial':
            p1[0] = p0[0] + 2.2 * ext
        elif ray == 'long':
            p0[0], p1[0] = -1.5 * ext, 1.5 * ext
        elif ray == 'rev':
            p0, p1 = p1[:], p0[:]
            p0[2], p1[2] = 40.0, 0.0
        elif ray == 'startbad':
            p0[2] = -100.0
        elif ray == 'vx_out':
            p0[0] = p1[0] = 3 * ext
        elif ray == 'hy_out':
            p0[1] = p1[1] = -3 * ext
        elif ray == 'before':
            p0[0], p1[0] = -4 * ext, -2 * ext
        elif ray == 'after':
            p0[0], p1[0] = 2 * ext, 4 * ext
        elif ray == 'tiny':
            p1 = [p0[0] + 1e-3, p0[1] - 2e-3, p0[2] - 1.0]
        mem.wu(RAY, 4, RES)
        mem.write(RAY + 0xC, struct.pack('<6f', *(p0 + p1)))
        mem.write(RES, bytes(16))
        mem.write(RES + 4, struct.pack('<f', 1.0))
        cpu.r['ecx'] = THIS
        return [RAY]
    return f


def stubs(case):
    hitp = case[4]

    def f(cpu, key, words, idx, seed):
        m = cpu.m
        if key == ('direct', GETH):
            p = m.read(words[0], 12)
            cpu.st.append(f32(10.0 + (sum(p) % 7)))
            return 4, {'p': p.hex()}
        if key in (('direct', CEIL), ('direct', FLOOR)):
            v = struct.unpack('<d', struct.pack('<II', words[0], words[1]))[0]
            cpu.st.append(float(math.ceil(v) if key[1] == CEIL else math.floor(v)))
            return 0, {'_cdecl': 1}, 2
        if key == ('direct', INVSQRT):
            v = struct.unpack('<f', struct.pack('<I', words[0]))[0]
            cpu.st.append(f32(1.0 / math.sqrt(v)) if v > 0 else math.inf)
            return 4, {'_cdecl': 1}, 1
        if key == ('direct', COLLIDE):
            seg = m.read(words[0], 24)
            ptrs = [m.u(words[1] + 4 * k, 4) for k in range(4)]
            tri = b''.join(m.read(p, 12) for p in ptrs)
            hit = hitp and (sum(tri) % hitp == 0)
            if hit:
                m.write(words[2] + 4, struct.pack('<f', 0.5))
            cpu.r['eax'] = (cpu.r['eax'] & ~0xFF) | (1 if hit else 0)
            return 0, {'_cdecl': 1, 'seg': seg.hex(), 'tri': tri.hex(), 'res': words[2]}, 0
        raise emu.Stop('callee %r' % (key,))
    return f


def main():
    H = Harness(OBJ, SYM, 0x4647C0, 1, void=False, retmask=0xFF, stk_any=True)
    if len(sys.argv) > 1 and sys.argv[1] == 'ctl':
        base = [(3, 1.0, 0, 'diag', 0), (4, 2.5, 0, 'long', 0), (3, 1.0, 1, 'diag', 7), (3, 1.0, 0, 'partial', 3)]
        H.run(base, setup, stubs, seeds=(1, 2), label='base')
        off, ins = H.find(lambda i: i.mnemonic == 'fmul' and i.op_str.endswith('+ 0x14]'))
        H.mutate('1 (+%#x %s %s: height scale -> y spacing)' % (off, ins.mnemonic, ins.op_str), off, ins.size - 1,
                 0x10, base, setup, stubs, seeds=(1, 2))
        off, ins = H.find(lambda i: i.mnemonic == 'mov' and i.op_str == 'cl, byte ptr [esi + 0x18]')
        H.mutate('2 (+%#x %s %s: wrap flag +0x18 -> +0x19)' % (off, ins.mnemonic, ins.op_str), off,
                 ins.size - 1, 0x19, base, setup, stubs, seeds=(1, 2))
        off, ins = H.find(lambda i: i.mnemonic == 'dec', 1)
        H.mutate('3 (+%#x %s %s: step count dec -> inc)' % (off, ins.mnemonic, ins.op_str), off, 0,
                 H.reader.code[off] - 8, base, setup, stubs, seeds=(1, 2))
        return
    H.run(CASES, setup, stubs, seeds=(1, 2), verbose=False)
    H.coverage()


if __name__ == '__main__':
    main()
