"""A13: ours vs the client's Unknown710530Class::Unknown_08 (the extrapolator's per-tick step) in emu.py."""
import struct
import sys
sys.path.insert(0, 'C:/EnBDevelopment/EnB/.work/a13')
import emu  # noqa: E402
from harness import Harness, fhash  # noqa: E402

OBJ = 'C:/EnBDevelopment/EnB/.work/a13/objs/extrap.obj'
SYM = '?Unknown_08@Unknown710530Class@@UAEXXZ'
THIS = 0x30000000
FLAGS = (0x40, 0x48, 0x58, 0x68, 0x7C, 0x88, 0x94)
MODES = ('mod', 'tiny', 'mix', 'big')
CASES = [(m, mode) for m in range(128) for mode in MODES]
TILTS = [(0.2, 0.5, 0.3), (1.5, 0.3, 2.0), (-1.5, -0.3, -2.0), (1.5, -0.3, 1.0), (-1.3, 0.2, -1.0), (1.0, 0.4, 1.5),
         (1.0, -0.4, -0.5), (-1.0, -0.5, -1.4), (0.0, 0.0, 0.0), (1.2217305, 0.1, 1.5)]
CASES += [(m, ('tilt', k)) for m in (0x04, 0x7F, 0x06) for k in range(len(TILTS))]


def val(seed, off, mode):
    v = fhash(seed, off) / 2  # [-2, 2)
    if mode == 'tiny':
        v *= 1e-6
    elif mode == 'mix':
        k = int(fhash(seed, off, 5) * 100) % 4
        v *= (1e-7, 1.0, 0.3, 1e-5)[k]
    elif mode == 'big':
        v *= 3.0
    return v


def setup(case):
    m, mode = case

    def f(mem, cpu, seed):
        for off in range(0x8, 0xA4, 4):
            if off in FLAGS:
                continue
            mem.write(THIS + off, struct.pack('<f', val(seed, off, mode if isinstance(mode, str) else 'mod')))
        if not isinstance(mode, str):
            mem.write(THIS + 0x5C, struct.pack('<3f', *TILTS[mode[1]]))
        for i, off in enumerate(FLAGS):
            mem.wu(THIS + off, 4, (m >> i) & 1)
        mem.wu(THIS + 0x40, 1, 1 if mode in ('mix', 'big', ('tilt', 1)) else 0)  # 0x40's flag from the mode (7 bits used)
        cpu.r['ecx'] = THIS
        return []
    return f


def stubs(case):
    def f(cpu, key, words, idx, seed):
        raise emu.Stop('callee %r' % (key,))
    return f


def main():
    H = Harness(OBJ, SYM, 0x710A20, 0, void=True)
    if len(sys.argv) > 1 and sys.argv[1] == 'ctl':
        base = [(127, 'mod'), (127, 'mix'), (0x55, 'big'), (0x2A, 'tiny'), (0x04, ('tilt', 1)), (0x7F, ('tilt', 0))]
        H.run(base, setup, stubs, seeds=(1, 2), label='base')
        off, ins = H.find(lambda i: i.mnemonic == 'fsin')
        H.mutate('1 (+%#x fsin -> fcos)' % off, off, 1, 0xFF, base, setup, stubs, seeds=(1, 2))
        off, ins = H.find(lambda i: i.mnemonic == 'cmp' and 'byte ptr' in i.op_str and '0x88' in i.op_str)
        H.mutate('2 (+%#x %s %s: flag 0x88 -> 0x89)' % (off, ins.mnemonic, ins.op_str), off, ins.size - 2, 0x89,
                 base, setup, stubs, seeds=(1, 2))
        off, ins = H.find(lambda i: i.mnemonic == 'fld' and i.op_str.endswith('+ 0x5c]'))
        H.mutate('3 (+%#x %s %s: tilt +0x5c -> +0x60)' % (off, ins.mnemonic, ins.op_str), off, ins.size - 1, 0x60,
                 base, setup, stubs, seeds=(1, 2))
        return
    H.run(CASES, setup, stubs, seeds=(1, 2), verbose=False)
    H.coverage()


if __name__ == '__main__':
    main()
