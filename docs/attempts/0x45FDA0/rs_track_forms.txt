"""Run authoritative object matching for every compiled form, without a similarity gate."""
import argparse
import collections
import json
from pathlib import Path
import subprocess
import sys

REPO = Path('/home/dev/Projects/EnB')
PYTHON = Path('/home/dev/Toolchains/enb-vc6-sp4/venv/bin/python')

def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('objects', type=Path, nargs='+')
    parser.add_argument('--out', type=Path, required=True)
    parser.add_argument('--target', type=lambda value: int(value, 0))
    arguments = parser.parse_args()
    arguments.out.mkdir(parents=True, exist_ok=True)
    summary = []
    failures = False
    for index, obj in enumerate(arguments.objects):
        label = f'{index:03d}_{obj.parent.name}'
        output = arguments.out / (label + '.json')
        command = [str(PYTHON), str(REPO / 'tools/re/objmatch.py'),
                   str(obj.resolve()), '-q', '--json', str(output.resolve())]
        process = subprocess.run(command, text=True, capture_output=True)
        (arguments.out / (label + '.log')).write_text(process.stdout + process.stderr)
        entry = {'object': str(obj), 'returncode': process.returncode}
        if process.returncode:
            failures = True
            entry['error'] = process.stderr.strip()
        else:
            rows = json.loads(output.read_text())
            entry['counts'] = dict(collections.Counter(row['status'] for row in rows))
            if arguments.target is not None:
                entry['target_exact'] = any(
                    row['status'] == 'exact' and
                    arguments.target in {int(address, 16) for address in row['addresses']}
                    for row in rows
                )
        summary.append(entry)
        print(json.dumps(entry), flush=True)
    (arguments.out / 'summary.json').write_text(json.dumps(summary, indent=2) + '\n')
    return int(failures)

if __name__ == '__main__':
    sys.exit(main())
