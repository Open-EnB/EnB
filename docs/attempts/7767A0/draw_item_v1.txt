#include <windows.h>
#include "..\..\wwlib\dsurface.h"
#include "inc3\unknown_htmlblock.h"
#include <string.h>
#include <stdio.h>
#include <ctype.h>

HFONT Get_Font(HDC hdc, char *face, int unknown, int height, int flags);

class HtmlImageClass
{
public:
	virtual void Unknown_00(void) = 0;
	virtual void Unknown_04(void) = 0;
	virtual Surface *Get_Surface(void) = 0;
	virtual char *Get_Name(void) = 0;
	virtual void Draw(Surface *surface, int x, int y, int clipx, int clipy, int clipw, int cliph) = 0;
};

struct HtmlLinkStruct
{
	ArrayList<char> Data;
	char Name[257];
	char Url[259];
};

class HtmlViewClass
{
public:
	bool Draw_Item(HtmlBlockClass *block, HtmlItemStruct *item, int offx, int offy, int clipx, int clipy, int clipw, int cliph);
	void Draw_Bevel(int x, int y, int width, int height, int clipx, int clipy, int clipw, int cliph, int thickness, bool raised);

	DSurface *Surf;
	void *Loader;
	ArrayList<HtmlBlockClass *> Blocks;
	ArrayList<HtmlLinkStruct> Links;
	ArrayList<HtmlImageClass *> Images;
};

// @0x7767A0
bool HtmlViewClass::Draw_Item(HtmlBlockClass *block, HtmlItemStruct *item, int offx, int offy, int clipx, int clipy, int clipw, int cliph)
{
	if (Surf == NULL)
		return false;

	char *type = item->Attributes.Get("Type");
	if (type == NULL)
		return true;

	if (strcmp(type, "Text") == 0) {
		char *text = item->Attributes.Get("Text");
		HDC hdc = Surf->GetDC();
		COLORREF color = 0;
		char *value = item->Attributes.Get("Font.Colorref");
		if (value != NULL || (value = block->Get_Attribute("TextColorref")) != NULL)
			color = atol(value);
		int flags = 0;
		if (item->Attributes.Get("Anchor.HREF") != NULL) {
			color = 0xB00000;
			char *link = block->Get_Attribute("LinkColorref");
			if (link != NULL)
				color = atol(link);
			flags = 4;
		}
		SetTextColor(hdc, color);
		SetBkMode(hdc, TRANSPARENT);
		if (item->Attributes.Get("Font.Bold") != NULL)
			flags |= 1;
		if (item->Attributes.Get("Font.Italic") != NULL)
			flags |= 2;
		int height = block->Get_Font_Height(item);
		char *face = item->Attributes.Get("Font.Face");
		if (face == NULL)
			face = "Times New Roman";
		HFONT font = Get_Font(hdc, face, 0, height - 2, flags);
		if (font != NULL)
			SelectObject(hdc, font);
		HRGN region = CreateRectRgn(clipx, clipy, clipw + clipx, clipy + cliph);
		SelectClipRgn(hdc, region);
		DeleteObject(region);

		int length = strlen(text);
		int y = item->Y - offy;
		int itemh = item->Height;
		int x = item->X - offx;
		int right = item->Width + x;
		int widest = 0;
		int dy = 0;
		while (length != 0) {
			SIZE size;
			int n = 1;
			while (true) {
				GetTextExtentPoint32(hdc, text, n, &size);
				if (size.cx > widest && size.cx + x < right)
					widest = size.cx;
				if (size.cx + x > right || n >= length)
					break;
				n++;
			}
			if (size.cx + x > right || n != length)
				n--;
			if (n < 1)
				n = 1;
			int found = -1;
			for (int k = n; k > 1 && n != length; k--) {
				if (!isgraph(text[k - 1])) {
					found = k;
					break;
				}
			}
			if (found != -1)
				n = found;
			SetTextAlign(hdc, TA_BASELINE);
			TextOut(hdc, x, dy + y - 2 + itemh, text, n);
			dy += height;
			text += n;
			length -= n;
		}
		Surf->ReleaseDC(hdc);
		return true;
	}

	if (strcmp(type, "HR") == 0) {
		Draw_Bevel(item->X - offx, item->Y - offy, item->Width, item->Height, clipx, clipy, clipw, cliph, 1, true);
		return true;
	}
	if (strcmp(type, "LI") == 0) {
		HDC hdc = Surf->GetDC();
		COLORREF color = 0;
		char *value = item->Attributes.Get("Font.Colorref");
		if (value != NULL || (value = block->Get_Attribute("TextColorref")) != NULL)
			color = atol(value);
		HBRUSH brush = CreateSolidBrush(color);
		RECT rect;
		rect.left = item->X - offx;
		rect.right = item->Width - offx + item->X;
		rect.top = item->Y - offy;
		rect.bottom = item->Height - offy + item->Y;
		FillRect(hdc, &rect, brush);
		DeleteObject(brush);
		Surf->ReleaseDC(hdc);
		return true;
	}
	if (strcmp(type, "Table") == 0) {
		char *border = item->Attributes.Get("Border");
		if (border != NULL) {
			Draw_Bevel(item->X - offx, item->Y - offy, item->Width, item->Height, clipx, clipy, clipw, cliph, atol(border), true);
			return true;
		}
	} else if (strcmp(type, "Cell") == 0) {
		char *border = item->Attributes.Get("Border");
		if (border != NULL) {
			Draw_Bevel(item->X - offx, item->Y - offy, item->Width, item->Height, clipx, clipy, clipw, cliph, atol(border), false);
			return true;
		}
	} else if (strcmp(type, "Image") == 0) {
		char *url = item->Attributes.Get("URL");
		for (int i = 0; i < Images.length(); i++) {
			HtmlImageClass *image;
			Images.get(image, i);
			if (strcmp(image->Get_Name(), url) == 0) {
				image->Draw(Surf, item->X - offx, item->Y - offy, clipx, clipy, clipw, cliph);
				return true;
			}
		}
	}
	return true;
}
