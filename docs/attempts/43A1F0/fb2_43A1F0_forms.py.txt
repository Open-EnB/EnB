def pad(text, n):
    return text + '\n' * (n - text.count('\n'))


def local(rows, last, n):
    out = '\tint result;\n'
    for i, (k, v) in enumerate(rows):
        out += '\t%sif (strcmp(name, "%s") == 0) { result = %s; }\n' % ('else ' if i else '', k, v)
    out += '\telse { result = (strcmp(name, "%s") == 0) ? %s : %s; }\n\treturn(result);\n' % last
    return pad(out, n)


def elseif(rows, last, n):
    out = ''
    for i, (k, v) in enumerate(rows):
        out += '\t%sif (strcmp(name, "%s") == 0) { return(%s); }\n' % ('else ' if i else '', k, v)
    out += '\telse { return((strcmp(name, "%s") == 0) ? %s : %s); }\n' % last

def lastif(rows, last, n):

    for k, v in rows:
        out += '\tif (strcmp(name, "%s") == 0) { return(%s); }\n' % (k, v)
    out += '\tif (strcmp(name, "%s") == 0) { return(%s); }\n\treturn(%s);\n' % last
    return pad(out, n)


def nested(rows, last, n):

    out += '(strcmp(name, "%s") == 0) ? %s : %s);\n' % last
    return pad(out, n)


FORMS = {}
for nm, f in (('local', local), ('elseif', elseif), ('lastif', lastif), ('nested', nested)):

"""Per form of a trybatch batch on 0x43A1F0: the rows of ours from the strncmp call to the first Get_String call
(the reloads of this / name and the registers they take), the row count and the real differing rows.

    python roles43.py BATCHDIR
"""
import os
import subprocess
import sys

ROOT = 'C:/EnBDevelopment/EnB/.work/wt/FB2'


def rows(obj):
    out = subprocess.run([sys.executable, ROOT + '/tools/re/objdiff.py', obj, 'MusicManagerClass::Load_INI', '0x43A1F0'],
                         capture_output=True, text=True).stdout.splitlines()
    res = []
    diff = 0
    for line in out[1:]:
        if '|' not in line:
            continue
        left, right = line.split('|', 1)
        l = left[1:].strip()
        lt = l.split(' ', 1)[1].strip() if ' ' in l else ''
        r = right.strip()
        rt = r.split(' ', 1)[1].strip() if ' ' in r else ''
        if left.startswith('*') and not ('REL' in rt and lt.split(' ')[0] == rt.split(' ')[0]):
            diff += 1
        if lt:
            res.append(lt)
    return res, diff


def main():
    batch = sys.argv[1]
    for form in sorted(os.listdir(batch)):
        obj = os.path.join(batch, form, 'obj', 'musicmanager.obj')
        if not os.path.exists(obj):
            continue
        rs, diff = rows(obj)
        seg = []
        for i, t in enumerate(rs):
            if t == 'push 0xa':
                j = i
                while j < len(rs) and not rs[j].startswith('jne'):
                    j += 1
                seg = [u for u in rs[j + 1:j + 12] if u.startswith('mov e') or u.startswith('push e')]
                break
        print('%-26s rows %d real-diff %-3d %s' % (form, len(rs), diff, ' ; '.join(seg[:6])))


main()
