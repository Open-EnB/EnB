"""A13: ours vs the client's LandPatchClass::Build_Vertices (a patch's vertex buffer from the height field) in emu.py."""
import struct
import sys
sys.path.insert(0, 'C:/EnBDevelopment/EnB/.work/a13')
import emu  # noqa: E402
from harness import Harness, fhash, f32  # noqa: E402

OBJ = 'C:/EnBDevelopment/EnB/.work/a13/objs/unknown_474530.obj'
SYM = '?Build_Vertices@LandPatchClass@@QAEXHHMM@Z'
THIS, VB, LAND, HF, VTX = 0x30000000, 0x31000000, 0x32000000, 0x33000000, 0x50000000
LOCK, UNLOCK, GETH = 0x94F030, 0x94F090, 0x463CD0
CASES = [(n, x0, y0, su, sv, hm) for n in (0, 1, 3, 5) for x0, y0 in ((0, 0), (7, -3)) for su, sv in ((1.0, 1.0),
         (0.5, 2.0)) for hm in ('hash', 'flat', 'neg', 'big')]


def setup(case):
    n, x0, y0, su, sv, hm = case

    def f(mem, cpu, seed):
        mem.wu(THIS, 4, VB)
        mem.wu(THIS + 4, 4, LAND)
        mem.wu(THIS + 0xC, 4, n)
        mem.wu(LAND + 0x90, 4, HF)
        mem.wu(HF + 0x20, 4, 64)
        mem.write(HF + 0x14, struct.pack('<f', 0.05))
        cpu.r['ecx'] = THIS
        return [x0 & 0xFFFFFFFF, y0 & 0xFFFFFFFF, struct.unpack('<I', struct.pack('<f', su))[0],
                struct.unpack('<I', struct.pack('<f', sv))[0]]
    return f


def stubs(case):
    hm = case[5]

    def f(cpu, key, words, idx, seed):
        m = cpu.m
        if key == ('direct', LOCK):
            m.wu(cpu.r['ecx'], 4, words[0])
            m.wu(cpu.r['ecx'] + 4, 4, VTX)
            return 4, None
        if key == ('direct', UNLOCK):
            return 0, None
        if key == ('direct', GETH):
            x, y = words[0], words[1]
            v = {'hash': fhash(seed, x, y) * 3, 'flat': 0.1, 'neg': -abs(fhash(seed, x, y)) * 3,
                 'big': 1e5 + fhash(seed, x, y)}[hm]
            cpu.st.append(f32(v))
            return 8, None
        raise emu.Stop('callee %r' % (key,))
    return f


def main():
    H = Harness(OBJ, SYM, 0x474790, 4, void=True, stk_any=True)
    if len(sys.argv) > 1 and sys.argv[1] == 'ctl':
        base = [(3, 7, -3, 0.5, 2.0, 'hash'), (5, 0, 0, 1.0, 1.0, 'hash')]
        H.run(base, setup, stubs, seeds=(1, 2), label='base')
        off, ins = H.find(lambda i: i.mnemonic == 'fmul' and i.op_str == 'dword ptr [esp + 0x64]')
        H.mutate('1 (+%#x %s %s: a scale arg -> the other)' % (off, ins.mnemonic, ins.op_str), off, ins.size - 1,
                 0x60, base, setup, stubs, seeds=(1, 2))
        off, ins = H.find(lambda i: i.mnemonic == 'mov' and i.op_str.endswith(', 0x40000000'))
        H.mutate('2 (+%#x %s %s: min start 2.0 -> 0.0)' % (off, ins.mnemonic, ins.op_str), off, ins.size - 1, 0,
                 base + [(3, 0, 0, 1.0, 1.0, 'flat')], setup, stubs, seeds=(1, 2))
        off, ins = H.find(lambda i: i.mnemonic == 'fstp' and i.op_str.endswith('+ 0x28]'))
        H.mutate('3 (+%#x %s %s: mid height +0x28 -> +0x2c)' % (off, ins.mnemonic, ins.op_str), off,
                 ins.size - 1, 0x2C, base, setup, stubs, seeds=(1, 2))
        return
    H.run(CASES, setup, stubs, seeds=(1, 2), verbose=False)
    H.coverage()


if __name__ == '__main__':
    main()
