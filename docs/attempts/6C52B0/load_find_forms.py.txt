def variant(find, store):
    body = load(find, store)
    new = ANCHOR.replace("}\n\n\nvoid FontStringFilesClass::Unload", "}" + body + "\n\nvoid FontStringFilesClass::Unload")

    assert SRC.count(blank_old) == 1
    blank_new = "\treturn -1;\n}\n" + "\n" * (64 - added)
    return [(C, ANCHOR, new), (C, blank_old, blank_new)]
