# ==== roles.py ====
"""Per form of a trybatch batch on 0x5C5DB0: the register this takes at entry, whether it is spilled at entry,
the register the loop's step takes, the row count and the count of direct calls (the inlining state), next to
trybatch's residue, so the register roles are read apart from the budget state.

    python roles.py BATCHDIR
"""
import os
import re
import subprocess
import sys

ROOT = 'C:/EnBDevelopment/EnB/.work/wt/FB2'


def rows(obj):
    out = subprocess.run([sys.executable, ROOT + '/tools/re/objdiff.py', obj, 'UIMissnClass::Unknown_5C5DB0', '0x5C5DB0'],
                         capture_output=True, text=True).stdout.splitlines()
    res = []
    diff = 0
    for line in out[1:]:
        if '|' not in line:
            continue
        left, right = line.split('|', 1)
        l = left[1:].strip()
        lt = l.split(' ', 1)[1].strip() if ' ' in l else ''
        r = right.strip()
        rt = r.split(' ', 1)[1].strip() if ' ' in r else ''
        if left.startswith('*') and not ('REL' in rt and lt.split(' ')[0] == rt.split(' ')[0]):
            diff += 1
        if lt:
            res.append(lt)
    return res, diff


def main():
    batch = sys.argv[1]
    seen = {}
    for form in sorted(os.listdir(batch)):
        obj = os.path.join(batch, form, 'obj', 'ui_missn.obj')
        if not os.path.exists(obj):
            continue
        rs, diff = rows(obj)
        this = next((t.split()[1].rstrip(',') for t in rs[:14] if re.match(r'mov e(bx|bp|si|di), ecx$', t)), '?')
        spill = any(re.match(r'mov dword ptr \[esp \+ 0xc\], e(bx|bp)$', t) for t in rs[:16])
        step = '?'
        for i, t in enumerate(rs):
            if t.startswith('cmp eax, 0x14'):
                for u in rs[i:i + 8]:
                    m = re.match(r'mov (e..), dword ptr \[e.. \+ 0x120\]', u)
                    if m:
                        step = m.group(1)
                break
        calls = sum(1 for t in rs if t == 'call REL')
        key = (this, spill, step, len(rs), calls, diff)
        print('%-28s this %s spill %-5s step %s rows %d calls %d real-diff %d' % ((form,) + key))


main()
